# English Coach PWA — Phase 1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A web app (PWA) for phone and desktop that listens to a meeting through one microphone, tells the learner's voice («Yo») from everyone else's («Interlocutor») by voice print, and shows the conversation transcribed and translated, with tap-to-correct and a manual "I'm speaking" mode.

**Architecture:** Plain ES modules served as static files, no build step. The microphone feeds an AudioWorklet and the energy `Segmenter` copied from the extension; each segment gets a speaker embedding (WeSpeaker ResNet34 through the vendored transformers.js, in a worker) that `speaker.js` compares with the learner's enrolled print and with a learnt centre of the other voice; the segment is transcribed by Groq and folded into sentence bubbles with the extension's `foldIntoTranscript`. Everything the learner owns (Groq key, voice print, last conversation) lives in IndexedDB on the device.

**Tech Stack:** HTML/CSS/JS modules, Web Audio (AudioWorklet), transformers.js 3 + ONNX Runtime Web (vendored), Groq HTTP API, IndexedDB, Wake Lock, Web App Manifest + service worker. Node 22+ for tooling and tests only; Google Chrome over the DevTools pipe for the end-to-end harness.

**Spec:** `docs/2026-09-29-plan.md` (Spanish, approved by the user on 2026-09-29, including "the PWA is for phone and desktop").

```js
// Who spoke, from speaker embeddings. The learner's print is enrolled once; the
// other voice has no print, so its centre is learnt from the turns assigned to it.

// Similarity to the learner's centre above which a turn is the learner's before
// any other voice is known (and, under 'hybrid', always). Calibrated by e2e/speakers.mjs.
export const MATCH_MIN = 0.5;
// The print outweighs any single turn, so one misassigned turn cannot drag the
// learner's centre toward the other voice.
export const PRINT_WEIGHT = 3;

export function normalize(v) {
  let n = 0;
  for (const x of v) n += x * x;
  n = Math.sqrt(n) || 1;
  return Float32Array.from(v, (x) => x / n);
}

export function cosine(a, b) {
  let s = 0;
  for (let i = 0; i < a.length; i++) s += a[i] * b[i];
  return s;
}

export function meanOf(vectors) {
  const out = new Float32Array(vectors[0].length);
  for (const v of vectors) {
    const u = normalize(v);
    for (let i = 0; i < u.length; i++) out[i] += u[i];
  }
  return normalize(out);
}

export function createSpeakers(print, { threshold = MATCH_MIN, rule = 'hybrid' } = {}) {
  return { print: normalize(print), threshold, rule, turns: new Map() };
}

function centreOf(state, who) {
  const vectors = [];
  for (const t of state.turns.values()) if (t.speaker === who) vectors.push(t.vector);
  if (who === 'me') return meanOf([...Array(PRINT_WEIGHT).fill(state.print), ...vectors]);
  return vectors.length ? meanOf(vectors) : null;
}

function decide(state, v) {
  const me = cosine(v, centreOf(state, 'me'));
  const centre = centreOf(state, 'them');
  const them = centre ? cosine(v, centre) : null;
  let speaker;
  if (them === null || state.rule === 'threshold') speaker = me >= state.threshold ? 'me' : 'them';
  else if (state.rule === 'centres') speaker = me >= them ? 'me' : 'them';
  else speaker = me >= state.threshold || me >= them ? 'me' : 'them';
  return { speaker, me, them };
}

export function classify(state, id, vector) {
  const v = normalize(vector);
  const result = decide(state, v);
  state.turns.set(id, { vector: v, speaker: result.speaker, manual: false });
  return result;
}

export function assign(state, id, vector, speaker) {
  state.turns.set(id, { vector: normalize(vector), speaker, manual: true });
}

export function correct(state, id, speaker) {
  const t = state.turns.get(id);
  if (!t) return false;
  t.speaker = speaker;
  t.manual = true;
  return true;
}

export function labelWindows(state, vectors) {
  return vectors.map((v) => decide(state, normalize(v)).speaker);
}
```

```js
// Pure DSP for synthetic meetings: what a phone on the table hears from a laptop
// speaker and from the learner sitting next to it. Deterministic by seed.

export function prng(seed) {
  let s = (seed >>> 0) || 1;
  return () => {
    s ^= s << 13; s >>>= 0;
    s ^= s >>> 17;
    s ^= s << 5; s >>>= 0;
    return s / 4294967296;
  };
}

// RBJ cookbook biquad.
export function biquad(x, type, f0, q, sr = 16000) {
  const w = 2 * Math.PI * f0 / sr;
  const c = Math.cos(w);
  const a = Math.sin(w) / (2 * q);
  const [b0, b1, b2] = type === 'lowpass' ? [(1 - c) / 2, 1 - c, (1 - c) / 2] : [(1 + c) / 2, -(1 + c), (1 + c) / 2];
  const a0 = 1 + a, a1 = -2 * c, a2 = 1 - a;
  const y = new Float32Array(x.length);
  let x1 = 0, x2 = 0, y1 = 0, y2 = 0;
  for (let n = 0; n < x.length; n++) {
    const yn = (b0 * x[n] + b1 * x1 + b2 * x2 - a1 * y1 - a2 * y2) / a0;
    y[n] = yn;
    x2 = x1; x1 = x[n]; y2 = y1; y1 = yn;
  }
  return y;
}

// Schroeder reverb: four parallel combs and two allpasses, linear time, which a
// convolution with a 0.3 s impulse over minutes of audio is not.
export function reverb(x, rt60, mixAmount, sr = 16000) {
  const combs = [29.7, 37.1, 41.1, 43.7].map((ms) => Math.round(ms * sr / 1000));
  const wet = new Float32Array(x.length);
  for (const d of combs) {
    const g = Math.pow(10, (-3 * d) / (rt60 * sr));
    const buf = new Float32Array(d);
    let i = 0;
    for (let n = 0; n < x.length; n++) {
      const out = buf[i];
      buf[i] = x[n] + out * g;
      wet[n] += out / combs.length;
      i = (i + 1) % d;
    }
  }
  let y = wet;
  for (const [ms, g] of [[5, 0.7], [1.7, 0.7]]) {
    const d = Math.round(ms * sr / 1000);
    const buf = new Float32Array(d);
    const out = new Float32Array(y.length);
    let i = 0;
    for (let n = 0; n < y.length; n++) {
      const delayed = buf[i];
      const v = y[n] + g * delayed;
      out[n] = delayed - g * v;
      buf[i] = v;
      i = (i + 1) % d;
    }
    y = out;
  }
  const res = new Float32Array(x.length);
  for (let n = 0; n < x.length; n++) res[n] = x[n] * (1 - mixAmount) + y[n] * mixAmount;
  return res;
}

export function noise(length, dbfs, seed) {
  const r = prng(seed);
  const amp = Math.pow(10, dbfs / 20);
  return biquad(Float32Array.from({ length }, () => (r() * 2 - 1) * amp * 1.7), 'lowpass', 1500, 0.707);
}

export function gainDb(x, db) {
  const g = Math.pow(10, db / 20);
  return Float32Array.from(x, (v) => v * g);
}

export function mix(a, b) {
  const out = new Float32Array(Math.max(a.length, b.length));
  for (let n = 0; n < out.length; n++) out[n] = (a[n] || 0) + (b[n] || 0);
  return out;
}

// A laptop speaker across the table: no bass, no air, a small resonance, the room
// around it, and 10 dB below a voice at arm's length.
export function laptopSpeaker(x, seed) {
  let y = biquad(x, 'highpass', 300, 0.707);
  y = biquad(y, 'lowpass', 6000, 0.707);
  y = reverb(y, 0.3, 0.35);
  return mix(gainDb(y, -10), noise(y.length, -60, seed));
}

// The learner at arm's length from the phone: mostly direct sound, a short room tail.
export function nearVoice(x, seed) {
  return mix(reverb(x, 0.15, 0.15), noise(x.length, -60, seed));
}
```
